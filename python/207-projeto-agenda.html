As views

# contatos/views.py
from django.contrib.auth.decorators import login_required
from django.db.models import Q
from django.shortcuts import get_object_or_404, redirect, render
from .forms import ContatoForm
from .models import Contato


@login_required
def lista(request):
    busca = request.GET.get("q", "")
    contatos = Contato.objects.filter(dono=request.user)
    if busca:
        contatos = contatos.filter(Q(nome__icontains=busca) | Q(telefone__icontains=busca))
    return render(request, "contatos/lista.html", {"contatos": contatos, "busca": busca})


@login_required
def criar(request):
    form = ContatoForm(request.POST or None)
    if form.is_valid():
        contato = form.save(commit=False)
        contato.dono = request.user
        contato.save()
        return redirect("lista")
    return render(request, "contatos/form.html", {"form": form, "titulo": "Novo contato"})


@login_required
def editar(request, pk):
    contato = get_object_or_404(Contato, pk=pk, dono=request.user)
    form = ContatoForm(request.POST or None, instance=contato)
    if form.is_valid():
        form.save()
        return redirect("lista")
    return render(request, "contatos/form.html", {"form": form, "titulo": "Editar contato"})


@login_required
def apagar(request, pk):
    contato = get_object_or_404(Contato, pk=pk, dono=request.user)
    if request.method == "POST":
        contato.delete()
        return redirect("lista")
    return render(request, "contatos/apagar.html", {"contato": contato})
  • instance=contato faz o formulário vir preenchido e atualizar o mesmo registro.
  • get_object_or_404 mostra a página "não encontrado" se o contato não existir ou for de outro usuário.
Atenção: Apague só com POST, depois de uma página de confirmação. Um link GET que apaga pode ser disparado sem querer (por exemplo, por um programa que pré-carrega links).

from django.contrib.auth.decorators import login_required
from django.shortcuts import get_object_or_404, render

@login_required
def detalhe(request, pk):
