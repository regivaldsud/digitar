O plano

CRUD quer dizer Create (criar), Read (ler), Update (atualizar) e Delete (apagar). Quase todo sistema é um ou vários CRUDs: agenda, biblioteca, estoque, lista de tarefas. Nesta aula você junta tudo das aulas 200 a 206 numa agenda completa, em que cada pessoa entra com a própria conta e só vê os próprios contatos.

Antes de programar, planeje as páginas. Cada linha vira uma URL e uma view:

  • / — lista os contatos, com busca (R);
  • /novo/ — formulário para criar (C);
  • /5/ — mostra o contato de número 5 (R);
  • /5/editar/ — formulário já preenchido (U);
  • /5/apagar/ — pergunta "tem certeza?" e apaga (D);
  • /contas/login/, /contas/logout/ e /cadastro/ — entrar, sair e criar conta (aula 206).

A lógica do CRUD, antes do Django

Por baixo, cada página faz um comando SQL (aulas 170 a 174). Veja as cinco operações da agenda com sqlite3, que roda aqui no navegador. Cada função é o "miolo" de uma view:

import sqlite3

con = sqlite3.connect(":memory:")
con.execute("""CREATE TABLE contato (
    id INTEGER PRIMARY KEY, dono TEXT, nome TEXT, telefone TEXT, favorito INTEGER DEFAULT 0)""")


def criar(dono, nome, telefone):
    cur = con.execute("INSERT INTO contato (dono, nome, telefone) VALUES (?, ?, ?)", (dono, nome, telefone))
    return cur.lastrowid


def lista(dono, busca=""):
    sql = "SELECT id, nome, telefone FROM contato WHERE dono = ? AND (nome LIKE ? OR telefone LIKE ?) ORDER BY favorito DESC, nome"
    return con.execute(sql, (dono, f"%{busca}%", f"%{busca}%")).fetchall()


def editar(dono, pk, nome, telefone, favorito):
    cur = con.execute("UPDATE contato SET nome = ?, telefone = ?, favorito = ? WHERE id = ? AND dono = ?",
                      (nome, telefone, favorito, pk, dono))
    return cur.rowcount == 1          # False = não achou (seria o 404)


def apagar(dono, pk):
    return con.execute("DELETE FROM contato WHERE id = ? AND dono = ?", (pk, dono)).rowcount == 1


criar("ana", "Bruno", "11 98888-1111")
caio = criar("ana", "Caio", "81 97777-2222")
criar("bia", "Dora", "71 96666-3333")
print("Lista da Ana:", lista("ana"))
print("Busca '81':", lista("ana", "81"))
print("Ana edita o Caio:", editar("ana", caio, "Caio Lima", "81 97777-2222", 1))
print("Bia tenta apagar o Caio:", apagar("bia", caio))
print("Lista da Ana:", lista("ana"))

Busca '81': [(2, 'Caio', '81 97777-2222')]

Repare no AND dono = ? em todas as operações: é ele que impede a Bia de mexer nos contatos da Ana. No Django, quem faz isso é o dono=request.user dentro do filter e do get_object_or_404.

Passo 3: o formulário

É o ContatoForm da aula 205. O campo dono não entra em fields: quem decide o dono é a view, nunca a pessoa que preenche.

# contatos/forms.py
from django import forms
from .models import Contato


class ContatoForm(forms.ModelForm):
    class Meta:
        model = Contato
        fields = ["nome", "telefone", "email", "nascimento", "favorito"]
        widgets = {"nascimento": forms.DateInput(attrs={"type": "date"})}

    def clean_nome(self):
        nome = self.cleaned_data["nome"].strip()
        if len(nome) < 2:
            raise forms.ValidationError("O nome precisa de pelo menos 2 letras.")
        return nome

    def clean_telefone(self):
        telefone = self.cleaned_data["telefone"]
        digitos = "".join(c for c in telefone if c.isdigit())
        if len(digitos) < 10:
            raise forms.ValidationError("Informe o DDD e o número.")
        return telefone

Passo 4: as views

Este é o coração da agenda. Leia com calma: cada view tem poucas linhas, e quase todas seguem o mesmo molde da aula 205.

# contatos/views.py
from django.contrib import messages
from django.contrib.auth import login
from django.contrib.auth.decorators import login_required
from django.contrib.auth.forms import UserCreationForm
from django.db.models import Q
from django.shortcuts import get_object_or_404, redirect, render
from .forms import ContatoForm
from .models import Contato


@login_required
def lista(request):
    busca = request.GET.get("q", "").strip()
    contatos = Contato.objects.filter(dono=request.user)
    if busca:
        contatos = contatos.filter(Q(nome__icontains=busca) | Q(telefone__icontains=busca))
    return render(request, "contatos/lista.html", {"contatos": contatos, "busca": busca})


@login_required
def detalhe(request, pk):
    contato = get_object_or_404(Contato, pk=pk, dono=request.user)
    return render(request, "contatos/detalhe.html", {"contato": contato})


@login_required
def criar(request):
    form = ContatoForm(request.POST or None)
    if form.is_valid():
        contato = form.save(commit=False)
        contato.dono = request.user
        contato.save()
        messages.success(request, f"{contato.nome} foi adicionado.")
        return redirect("lista")
    return render(request, "contatos/form.html", {"form": form, "titulo": "Novo contato"})


@login_required
def editar(request, pk):
    contato = get_object_or_404(Contato, pk=pk, dono=request.user)
    form = ContatoForm(request.POST or None, instance=contato)
    if form.is_valid():
        form.save()
        messages.success(request, "Contato atualizado.")
        return redirect("detalhe", pk=contato.pk)
    return render(request, "contatos/form.html", {"form": form, "titulo": "Editar contato"})


@login_required
def apagar(request, pk):
    contato = get_object_or_404(Contato, pk=pk, dono=request.user)
    if request.method == "POST":
        contato.delete()
        messages.success(request, f"{contato.nome} foi apagado.")
        return redirect("lista")
    return render(request, "contatos/apagar.html", {"contato": contato})


def cadastro(request):
    form = UserCreationForm(request.POST or None)
    if form.is_valid():
        usuario = form.save()
        login(request, usuario)
        return redirect("lista")
    return render(request, "registration/cadastro.html", {"form": form})
  • get_object_or_404(Contato, pk=pk, dono=request.user) devolve a página "não encontrado" (404) se o contato não existir ou for de outra pessoa.
  • instance=contato faz o formulário vir preenchido e, no save(), atualizar o mesmo registro em vez de criar outro.
  • messages.success guarda um recado que aparece uma vez só, na próxima página.
Atenção: Apague só com POST, depois de uma página de confirmação. Se um simples link (GET) apagasse, um clique sem querer — ou um programa que pré-carrega links — sumiria com os seus dados. E esquecer o dono=request.user no get_object_or_404 deixaria qualquer pessoa logada editar o contato dos outros só trocando o número no endereço.

Passo 8: testar tudo automaticamente

Clicar em tudo a cada mudança cansa. O Django tem um navegador de mentira, o Client, que faz pedidos sem abrir o navegador (testes como na aula 131). Salve em contatos/tests.py e rode python manage.py test:

# contatos/tests.py
from django.contrib.auth.models import User
from django.test import Client, TestCase
from .models import Contato


class AgendaTest(TestCase):
    def setUp(self):
        User.objects.create_user("ana", password="senha-forte-123")
        User.objects.create_user("bia", password="senha-forte-123")
        self.client.login(username="ana", password="senha-forte-123")

    def test_precisa_de_login(self):
        resposta = Client().get("/")
        self.assertRedirects(resposta, "/contas/login/?next=/")

    def test_criar_editar_apagar(self):
        self.client.post("/novo/", {"nome": "Bruno", "telefone": "11 98888-1111"})
        bruno = Contato.objects.get(nome="Bruno")
        self.assertContains(self.client.get("/"), "Bruno")
        self.client.post(f"/{bruno.pk}/editar/", {"nome": "Bruno Lima", "telefone": "11 98888-1111"})
        bruno.refresh_from_db()
        self.assertEqual(bruno.nome, "Bruno Lima")
        self.client.post(f"/{bruno.pk}/apagar/")
        self.assertEqual(Contato.objects.count(), 0)

    def test_outra_pessoa_nao_ve(self):
        self.client.post("/novo/", {"nome": "Bruno", "telefone": "11 98888-1111"})
        bruno = Contato.objects.get(nome="Bruno")
        bia = Client()
        bia.login(username="bia", password="senha-forte-123")
        self.assertEqual(bia.get(f"/{bruno.pk}/").status_code, 404)

    def test_post_sem_csrf_e_recusado(self):
        protegido = Client(enforce_csrf_checks=True)
        protegido.login(username="ana", password="senha-forte-123")
        resposta = protegido.post("/novo/", {"nome": "Caio", "telefone": "11 97777-0000"})
        self.assertEqual(resposta.status_code, 403)

# contatos/views.py
from django.contrib.auth.decorators import login_required
from django.shortcuts import get_object_or_404, redirect
from django.views.decorators.http import require_POST
from .models import Contato


@login_required
@require_POST
def favoritar(request, pk):

5. Por que a view editar usa ContatoForm(request.POST or None, instance=contato) e não só ContatoForm(request.POST)? O que aconteceria sem o instance?

Ver resposta

O instance diz ao formulário qual contato ele está editando:
no GET ele aparece preenchido com os dados atuais, e no save() o Django faz um UPDATE nesse registro.
Sem o instance, o formulário viria vazio e o save() criaria um contato NOVO (um INSERT),
deixando o antigo igual e a agenda com dois contatos.
