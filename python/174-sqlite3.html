Conexão e cursor

A conexão (connect) representa o banco. O cursor é quem executa os comandos e percorre os resultados. O con.execute que usamos até aqui é um atalho que cria um cursor para você.

import sqlite3

con = sqlite3.connect(":memory:")
cur = con.cursor()
cur.execute("CREATE TABLE livros (id INTEGER PRIMARY KEY, titulo TEXT, autor TEXT)")
cur.execute("INSERT INTO livros (titulo, autor) VALUES (?, ?)", ("Dom Casmurro", "Machado de Assis"))
print("id do novo livro:", cur.lastrowid)
cur.execute("SELECT titulo, autor FROM livros")
print(cur.fetchone())
print(cur.fetchone())
con.close()

id do novo livro: 1
('Dom Casmurro', 'Machado de Assis')
None

lastrowid diz o id gerado no último INSERT. Quando não há mais linhas, fetchone() devolve None.

CRUD em funções

import sqlite3


def criar_banco():
    con = sqlite3.connect(":memory:")
    con.execute("CREATE TABLE livros (id INTEGER PRIMARY KEY, titulo TEXT, autor TEXT)")
    return con


def adicionar(con, titulo, autor):
    with con:
        cur = con.execute("INSERT INTO livros (titulo, autor) VALUES (?, ?)", (titulo, autor))
    return cur.lastrowid


def listar(con):
    return con.execute("SELECT id, titulo, autor FROM livros ORDER BY titulo").fetchall()


def trocar_autor(con, id_livro, autor):
    with con:
        con.execute("UPDATE livros SET autor = ? WHERE id = ?", (autor, id_livro))


def remover(con, id_livro):
    with con:
        con.execute("DELETE FROM livros WHERE id = ?", (id_livro,))


con = criar_banco()
adicionar(con, "O Cortiço", "Aluísio Azevedo")
n = adicionar(con, "Vidas Secas", "Graciliano")
adicionar(con, "Iracema", "José de Alencar")
trocar_autor(con, n, "Graciliano Ramos")
remover(con, 1)
for livro in listar(con):
    print(livro)
con.close()

(3, 'Iracema', 'José de Alencar')
(2, 'Vidas Secas', 'Graciliano Ramos')

Atenção: Uma tupla de um item só precisa de vírgula: (id_livro,). Sem a vírgula, (id_livro) é só um número entre parênteses e o sqlite3 reclama.

Linhas como dicionários

Com row_factory = sqlite3.Row, cada linha pode ser lida pelo nome da coluna:

import sqlite3

con = sqlite3.connect(":memory:")
con.row_factory = sqlite3.Row
con.execute("CREATE TABLE times (nome TEXT, estado TEXT, titulos INTEGER)")
con.executemany("INSERT INTO times VALUES (?, ?, ?)",
                [("Bahia", "BA", 2), ("Fortaleza", "CE", 0)])
for t in con.execute("SELECT * FROM times"):
    print(t["nome"], "-", t["estado"], "-", t["titulos"], "título(s) brasileiro(s)")
con.close()

Bahia - BA - 2 título(s) brasileiro(s)
Fortaleza - CE - 0 título(s) brasileiro(s)

Dica: Quando você roda este código no seu computador com sqlite3.connect("biblioteca.db"), o arquivo biblioteca.db aparece na pasta. Programas como o DB Browser for SQLite abrem esse arquivo para você ver as tabelas.
Pratique no DIGITAR: digite este trecho no modo Programar do app. 
cur.execute("SELECT * FROM livros WHERE id = ?", (1,))

import sqlite3


def buscar_por_autor(con, trecho):
    sql = "SELECT titulo FROM livros WHERE autor LIKE ? ORDER BY titulo"
    return [linha[0] for linha in con.execute(sql, (f"%{trecho}%",))]


con = sqlite3.connect(":memory:")

def contar_livros(con):
    return con.execute("SELECT COUNT(*) FROM livros").fetchone()[0]
