Os models

# loja/models.py
from django.conf import settings
from django.db import models


class Produto(models.Model):
    nome = models.CharField(max_length=120)
    slug = models.SlugField(unique=True)
    descricao = models.TextField(blank=True)

    def __str__(self):
        return self.nome


class Variacao(models.Model):
    produto = models.ForeignKey(Produto, on_delete=models.CASCADE, related_name="variacoes")
    nome = models.CharField(max_length=60)  # ex.: "Camiseta P azul"
    preco = models.DecimalField(max_digits=8, decimal_places=2)
    estoque = models.PositiveIntegerField(default=0)

    def __str__(self):
        return f"{self.produto} - {self.nome}"


class Pedido(models.Model):
    usuario = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.PROTECT)
    total = models.DecimalField(max_digits=10, decimal_places=2)
    criado_em = models.DateTimeField(auto_now_add=True)


class ItemPedido(models.Model):
    pedido = models.ForeignKey(Pedido, on_delete=models.CASCADE, related_name="itens")
    variacao = models.ForeignKey(Variacao, on_delete=models.PROTECT)
    quantidade = models.PositiveIntegerField()
    preco = models.DecimalField(max_digits=8, decimal_places=2)
Atenção: Dinheiro sempre com DecimalField e Decimal (aula 29), nunca float: 0,1 + 0,2 não dá exatamente 0,3 com float.

O ItemPedido guarda o preço do momento da compra: se o produto ficar mais caro amanhã, o pedido antigo continua certo.

O carrinho na sessão

A sessão é um dicionário que o Django guarda para cada visitante, mesmo sem login. O carrinho pode ser {id_da_variacao: quantidade}. Como a sessão é salva em JSON, as chaves viram texto:

from django.shortcuts import get_object_or_404, redirect
from .models import Variacao


def adicionar(request, variacao_id):
    variacao = get_object_or_404(Variacao, pk=variacao_id)
    carrinho = request.session.get("carrinho", {})
    chave = str(variacao.pk)
    carrinho[chave] = min(carrinho.get(chave, 0) + 1, variacao.estoque)
    request.session["carrinho"] = carrinho
    return redirect("carrinho")

A conta do carrinho, em Python puro

from decimal import Decimal

precos = {"1": Decimal("49.90"), "2": Decimal("12.50")}
carrinho = {"1": 2, "2": 3}

total = sum(precos[chave] * qtd for chave, qtd in carrinho.items())
print(f"Total: R$ {total}")

Exercícios

1. Escreva a função aplicar_cupom(total, porcento) com Decimal que devolve o total com desconto, arredondado a 2 casas. Teste com R$ 137,30 e 10%.

Ver resposta 
from decimal import Decimal, ROUND_HALF_UP


def aplicar_cupom(total, porcento):
    desconto = total * Decimal(porcento) / 100
    return (total - desconto).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)


print(aplicar_cupom(Decimal("137.30"), 10))

123.57

2. Escreva a view remover(request, variacao_id) que tira o item do carrinho da sessão.

Ver resposta 
from django.shortcuts import redirect


def remover(request, variacao_id):
    carrinho = request.session.get("carrinho", {})
    carrinho.pop(str(variacao_id), None)
    request.session["carrinho"] = carrinho
    return redirect("carrinho")

pop(chave, None) não dá erro se o item não estiver no carrinho.
