Os models

# blog/models.py
from django.db import models
from django.urls import reverse
from django.utils.text import slugify


class Categoria(models.Model):
    nome = models.CharField(max_length=60)
    slug = models.SlugField(unique=True)

    def __str__(self):
        return self.nome


class Tag(models.Model):
    nome = models.CharField(max_length=40, unique=True)

    def __str__(self):
        return self.nome


class Post(models.Model):
    titulo = models.CharField(max_length=150)
    slug = models.SlugField(unique=True, blank=True)
    categoria = models.ForeignKey(Categoria, on_delete=models.PROTECT, related_name="posts")
    tags = models.ManyToManyField(Tag, blank=True)
    capa = models.ImageField(upload_to="capas/", blank=True)
    texto = models.TextField()
    publicado = models.BooleanField(default=False)
    criado_em = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["-criado_em"]

    def save(self, *args, **kwargs):
        if not self.slug:
            self.slug = slugify(self.titulo)
        super().save(*args, **kwargs)

    def get_absolute_url(self):
        return reverse("post", args=[self.slug])

    def __str__(self):
        return self.titulo
  • Uma categoria tem muitos posts (ForeignKey); um post tem muitas tags, e uma tag está em muitos posts (ManyToManyField). Veja JOIN e chaves na aula 173.
  • O ImageField precisa do pacote Pillow: pip install pillow.
Atenção: slugify remove acentos e espaços: "Pão de queijo mineiro" vira pao-de-queijo-mineiro. Dois posts com o mesmo título dariam o mesmo slug e um erro de unique; num blog real, acrescente um número ao final quando já existir.

A ideia do slug, em Python puro

import re
import unicodedata


def meu_slugify(texto):
    sem_acento = unicodedata.normalize("NFKD", texto).encode("ascii", "ignore").decode()
    sem_acento = re.sub(r"[^a-z0-9]+", "-", sem_acento.lower())
    return sem_acento.strip("-")


print(meu_slugify("Pão de queijo mineiro!"))
print(meu_slugify("10 dicas de digitação"))

pao-de-queijo-mineiro
10-dicas-de-digitacao

from .models import Categoria


def categorias(request):
    return {"menu_categorias": Categoria.objects.all()}
# config/urls.py (trecho, só para desenvolvimento)
from django.conf import settings
from django.conf.urls.static import static
from django.urls import include, path

urlpatterns = [
    path("", include("blog.urls")),
] + static(settings.MEDIA_URL, document_root=settings.MEDIA_ROOT)

Um context processor é uma função que acrescenta variáveis ao contexto de todos os templates. Assim o base.html pode listar menu_categorias sem cada view ter que mandar.

Dica: Num formulário com upload, use <form method="post" enctype="multipart/form-data"> e passe request.FILES para o form. Sem isso, a imagem não chega.

Exercícios

1. Escreva a view post(request, slug) que mostra só posts publicados e a URL correspondente.

Ver resposta 
from django.shortcuts import get_object_or_404, render
from .models import Post


def post(request, slug):
    item = get_object_or_404(Post, slug=slug, publicado=True)
    return render(request, "blog/post.html", {"post": item})
