Passo 1: a lógica, sem janela

Um bom hábito: escreva a parte que calcula separada da tela. Assim você testa sem abrir janela nenhuma. A função abaixo recebe a conta como texto e devolve o resultado.

def calcular(expressao):
    permitido = "0123456789+-*/.() "
    if not expressao or any(c not in permitido for c in expressao):
        raise ValueError("Conta inválida")
    resultado = eval(expressao, {"__builtins__": {}})
    if resultado == int(resultado):
        resultado = int(resultado)
    return resultado


print(calcular("12+30"))
print(calcular("7/2"))
print(calcular("(2+3)*4"))

42
3.5
20

Atenção: eval roda qualquer código Python, por isso é perigoso com texto vindo do usuário. Aqui só deixamos passar dígitos e operadores, e tiramos as funções embutidas. Num programa sério, use um analisador de expressões.

Passo 2: a janela da calculadora

import sys
from PySide6.QtCore import Qt
from PySide6.QtWidgets import (QApplication, QWidget, QGridLayout,
                               QLineEdit, QPushButton, QMessageBox)

BOTOES = [
    ["7", "8", "9", "/"],
    ["4", "5", "6", "*"],
    ["1", "2", "3", "-"],
    ["C", "0", "=", "+"],
]


def calcular(expressao):
    permitido = "0123456789+-*/.() "
    if not expressao or any(c not in permitido for c in expressao):
        raise ValueError("Conta inválida")
    resultado = eval(expressao, {"__builtins__": {}})
    if resultado == int(resultado):
        resultado = int(resultado)
    return resultado


class Calculadora(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Calculadora")
        self.display = QLineEdit()
        self.display.setReadOnly(True)
        self.display.setAlignment(Qt.AlignmentFlag.AlignRight)
        grade = QGridLayout(self)
        grade.addWidget(self.display, 0, 0, 1, 4)
        for linha, textos in enumerate(BOTOES, start=1):
            for coluna, texto in enumerate(textos):
                botao = QPushButton(texto)
                botao.clicked.connect(lambda _=False, t=texto: self.apertar(t))
                grade.addWidget(botao, linha, coluna)

    def apertar(self, texto):
        if texto == "C":
            self.display.clear()
        elif texto == "=":
            try:
                self.display.setText(str(calcular(self.display.text())))
            except (ValueError, SyntaxError, ZeroDivisionError):
                QMessageBox.warning(self, "Ops", "Essa conta não dá para fazer.")
                self.display.clear()
        else:
            self.display.setText(self.display.text() + texto)

    def keyPressEvent(self, evento):
        tecla = evento.text()
        if tecla in "0123456789+-*/":
            self.apertar(tecla)
        elif evento.key() in (Qt.Key.Key_Return, Qt.Key.Key_Enter):
            self.apertar("=")
        elif evento.key() == Qt.Key.Key_Escape:
            self.apertar("C")
        else:
            super().keyPressEvent(evento)


app = QApplication(sys.argv)
janela = Calculadora()
janela.show()
sys.exit(app.exec())

Pontos importantes:

  • grade.addWidget(self.display, 0, 0, 1, 4) faz o display ocupar 1 linha e 4 colunas.
  • Na lambda, t=texto guarda o texto de cada botão. Sem isso, todos os botões usariam o último texto do laço.
  • keyPressEvent é chamado a cada tecla: assim dá para usar a calculadora pelo teclado — ótimo para quem treina digitação!
Dica: Teste com 5/0: o ZeroDivisionError é capturado e aparece a caixa de aviso do QMessageBox.

2. Escreva a função apagar_ultimo(texto) que será usada pelo botão "⌫" e teste com "123+4".

Ver resposta 
def apagar_ultimo(texto):
    return texto[:-1]


print(apagar_ultimo("123+4"))
print(repr(apagar_ultimo("")))

123+
''

Fatiar [:-1] um texto vazio devolve outro texto vazio, então não há erro.
